def move_to_position(position, rotation):
    """
    Move the end-effector to a specified position with a given rotation.
    
    Parameters:
    - position: List containing x, y, z coordinates [x, y, z]
    - rotation: Rotation value in radians
    """
    # Here, we would define the trajectory points for the movement
    # to the desired position. For simplicity, we assume a linear trajectory.
    trajectory = []
    start_position = [0.0, 0.6, 0.55]
    
    # Generate intermediate points
    for t in range(100):
        intermediate_x = start_position[0] + (position[0] - start_position[0]) * t / 100
        intermediate_y = start_position[1] + (position[1] - start_position[1]) * t / 100
        intermediate_z = start_position[2] + (position[2] - start_position[2]) * t / 100
        trajectory.append([intermediate_x, intermediate_y, intermediate_z, rotation])
    
    return trajectory

def trajectory_to_above(position, rotation):
    trajectory = []
    start_x, start_y, start_z = 0.0, 0.6, 0.55  # Start position
    end_x, end_y, end_z = position  # Target position above the object
    num_points = 100

    for i in range(num_points + 1):
        t = i / num_points
        x = start_x + t * (end_x - start_x)
        y = start_y + t * (end_y - start_y)
        z = start_z + t * (end_z - start_z)
        trajectory.append([x, y, z, rotation])
    
    return trajectory
